function tmdbFetch(path) {
  return fetch(PROXY_URL + path)
    .then(r => { if (!r.ok) throw new Error('HTTP ' + r.status); return r.json(); });
}

function normalizeItem(r) {
  const type = r.media_type || (r.first_air_date ? 'tv' : 'movie');
  return {
    key: type + '-' + r.id, id: r.id, type,
    title: r.title || r.name, year: (r.release_date || r.first_air_date || '').slice(0, 4),
    poster: r.poster_path, 
    backdrop: r.backdrop_path, // <--- ДОДАЙ ЦЕ
    overview: r.overview || '', tmdbRating: r.vote_average || 0,
    votes: r.vote_count || 0, popularity: r.popularity || 0,
    genres: r.genre_ids || (r.genres ? r.genres.map(g => g.id) : [])
  };
}

function findAllTrailers(videos, originalLang) {
  const list = (videos?.results || []).filter(v => v.site === 'YouTube' && v.key && (v.type === 'Trailer' || v.type === 'Teaser'));
  const langRank = v => v.iso_639_1 === 'uk' ? 0 : (v.iso_639_1 === originalLang ? 1 : (v.iso_639_1 === 'en' ? 2 : 3));
  const typeRank = v => v.type === 'Trailer' ? 0 : 1;
  const seen = new Set();
  return list.sort((a, b) =>
      typeRank(a) - typeRank(b) || langRank(a) - langRank(b) ||
      (b.official ? 1 : 0) - (a.official ? 1 : 0) || new Date(b.published_at) - new Date(a.published_at))
    .filter(v => { if (seen.has(v.key)) return false; seen.add(v.key); return true; }).slice(0, 5);
}

function trailerLabel(t) {
  const lang = t.iso_639_1 === 'uk' ? 'UA' : (t.iso_639_1 ? t.iso_639_1.toUpperCase() : '—');
  return (t.type === 'Teaser' ? 'Тізер' : 'Трейлер') + ' · ' + lang;
}

function setTab(t) {
  state.tab = t;
  closeSidebar();
  document.querySelectorAll(".nav-item").forEach(b => b.classList.toggle("active", b.dataset.tab === t));
  typeFilterBar.classList.toggle("hidden", !["recent", "watching", "watched", "planned", "dropped", "favorites"].includes(t));
  if (t === 'game') { box.style.padding = '0'; renderGame(); return; }
  if (t === 'top') { box.style.padding = '0 16px 16px'; renderMyTop(); return; }
  refresh();
}

function renderCalendar() {
  box.style.padding = '0';
  box.replaceChildren();
  
  const wrap = el('div', 'home-wrap');
  const head = el('div', 'section-head');
  head.style.marginTop = '20px';
  head.append(el('h1', 'section-title', '📅 Календар'));
  wrap.append(head);

  // 1. Беремо всі тайтли з потрібними статусами
  let allItems = Object.values(lib).filter(x => x.status === 'watching' || x.status === 'watched');
  
  if (!allItems.length) {
    wrap.append(el('div', 'avail-empty', 'Тут поки порожньо. Додай фільми зі статусом "Дивлюсь" або "Переглянуто".'));
    box.append(wrap);
    return;
  }

  // 2. Фільтруємо за обраним статусом (якщо не 'all')
  let items = allItems;
  if (state.calendarStatus === 'watching') items = items.filter(x => x.status === 'watching');
  else if (state.calendarStatus === 'watched') items = items.filter(x => x.status === 'watched');

  // 3. Збираємо унікальні місяці вже з відфільтрованих елементів
  const monthsSet = new Set();
  items.forEach(item => {
    const date = new Date(item.statusDate || item.updated || Date.now());
    const monthKey = `${date.getFullYear()}-${date.getMonth()}`;
    monthsSet.add(monthKey);
  });
  
  const months = Array.from(monthsSet).sort((a, b) => {
    const [ya, ma] = a.split('-').map(Number);
    const [yb, mb] = b.split('-').map(Number);
    return yb - ya || mb - ma;
  });

  // Якщо поточний вибраний місяць відсутній у новому списку - беремо перший
  if (!state.calendarMonth || !monthsSet.has(state.calendarMonth)) {
    state.calendarMonth = months[0] || null;
  }

  // 4. Малюємо панель з фільтрами (Місяць + Статус)
  const calHeader = el('div', 'calendar-header');
  calHeader.style.display = 'flex';
  calHeader.style.gap = '10px';
  calHeader.style.flexWrap = 'wrap';

  // Селект статусу
  const sSelect = document.createElement('select');
  sSelect.className = 'calendar-select';
  sSelect.style.flex = '1';
  sSelect.style.minWidth = '150px';
  [
    { val: 'all', label: 'Всі статуси' },
    { val: 'watching', label: '▶️ Дивлюсь' },
    { val: 'watched', label: '✅ Переглянуто' }
  ].forEach(optData => {
    const opt = document.createElement('option');
    opt.value = optData.val;
    opt.textContent = optData.label;
    if (state.calendarStatus === optData.val) opt.selected = true;
    sSelect.append(opt);
  });
  sSelect.addEventListener('change', (e) => {
    state.calendarStatus = e.target.value;
    state.calendarMonth = null; // Скидаємо місяць, щоб він обрався автоматично з нового списку
    renderCalendar();
  });
  calHeader.append(sSelect);

  // Селект місяця
  if (months.length > 0) {
    const mSelect = document.createElement('select');
    mSelect.className = 'calendar-select';
    mSelect.style.flex = '1';
    mSelect.style.minWidth = '150px';
    
    const monthNames = ["Січень", "Лютий", "Березень", "Квітень", "Травень", "Червень", "Липень", "Серпень", "Вересень", "Жовтень", "Листопад", "Грудень"];
    
    months.forEach(m => {
      const [y, monthIdx] = m.split('-').map(Number);
      const opt = document.createElement('option');
      opt.value = m;
      opt.textContent = `${monthNames[monthIdx]} ${y}`;
      if (m === state.calendarMonth) opt.selected = true;
      mSelect.append(opt);
    });
    
    mSelect.addEventListener('change', (e) => {
      state.calendarMonth = e.target.value;
      renderCalendar();
    });
    calHeader.append(mSelect);
  }
  
  wrap.append(calHeader);

  if (!items.length) {
    wrap.append(el('div', 'avail-empty', 'Немає записів з таким статусом.'));
    box.append(wrap);
    return;
  }

  // 5. Фільтруємо по вибраному місяцю
  const filteredItems = items.filter(item => {
    const date = new Date(item.statusDate || item.updated || Date.now());
    return `${date.getFullYear()}-${date.getMonth()}` === state.calendarMonth;
  });

  // 6. Групуємо за днями
  const groups = {};
  filteredItems.forEach(item => {
    const date = new Date(item.statusDate || item.updated || Date.now());
    const dayKey = date.toLocaleDateString('uk-UA', { day: 'numeric', month: 'long', year: 'numeric' });
    if (!groups[dayKey]) groups[dayKey] = [];
    groups[dayKey].push(item);
  });

  // Сортуємо дні за спаданням
  const sortedDays = Object.keys(groups).sort((a, b) => {
    return new Date(groups[b][0].statusDate || groups[b][0].updated) - new Date(groups[a][0].statusDate || groups[a][0].updated);
  });

  const timeline = el('div', 'timeline');
  
  sortedDays.forEach(day => {
    const dayGroup = el('div', 'timeline-day-group');
    dayGroup.append(el('div', 'timeline-date', day));
    
    const dayItems = el('div', 'timeline-items');
    
    groups[day].sort((a, b) => (b.statusDate || b.updated) - (a.statusDate || a.updated)).forEach(item => {
      const listItem = createListItem(item);
      
      const statusBadge = item.status === 'watching' ? '▶️ Дивлюсь' : '✅ Переглянуто';
      const info = listItem.querySelector('.info');
      if (info) {
        const badge = el('span', 'badge', statusBadge);
        badge.style.background = item.status === 'watching' ? 'rgba(var(--accent-rgb), 0.15)' : 'rgba(157, 229, 195, 0.15)';
        badge.style.color = item.status === 'watching' ? 'var(--accent)' : 'var(--mint)';
        info.insertBefore(badge, info.firstChild);
      }
      
      dayItems.append(listItem);
    });
    
    dayGroup.append(dayItems);
    timeline.append(dayGroup);
  });

  wrap.append(timeline);
  box.append(wrap);
}

function hashFor(ctx) { return ctx.type === 'person' ? `#/person/${ctx.id}` : `#/${ctx.item.type}/${ctx.item.id}`; }
function pushCardState() {
  const ctx = cardStack[cardStack.length - 1];
  if (!ctx) return;
  const st = { mc: cardStack.length };
  if (ctx.type === 'person') st.ctx = { type: 'person', id: ctx.id };
  else st.ctx = { type: 'media', id: ctx.item.id, media_type: ctx.item.type, key: ctx.item.key };
  history.pushState(st, '', hashFor(ctx));
}
function saveCardScroll() {
  const s = card.querySelector('.card-scroll');
  const top = cardStack[cardStack.length - 1];
  if (s && top) top.scroll = s.scrollTop;
}

async function openCard(r) {
  if (cardStack.length) return openMediaFromStack(r);
  overlay.classList.add("open");
  const ctx = { type: 'media', item: r, activeTab: 'cast', seasonState: {}, playingTrailer: false, overviewExpanded: false, logoPath: null };
  cardStack = [ctx];
  pushCardState();
  renderLoading();
  await loadMediaCardData(ctx);
  if (cardStack[cardStack.length - 1] !== ctx) return;
  renderCard();
}

async function openMediaFromStack(item) {
  saveCardScroll();
  const ctx = { type: 'media', item, activeTab: 'cast', seasonState: {}, playingTrailer: false, overviewExpanded: false, logoPath: null };
  cardStack.push(ctx);
  pushCardState();
  renderLoading();
  await loadMediaCardData(ctx);
  if (cardStack[cardStack.length - 1] !== ctx) return;
  renderCard();
}

function cardBar() {
  const bar = el('div', 'card-bar');
  const back = el('button', 'card-x', '←'); back.title = 'Назад';
  back.addEventListener('click', () => history.back());
  bar.append(back);
  if (cardStack.length > 1) {
    const x = el('button', 'card-x', '✕'); x.title = 'Закрити все';
    x.addEventListener('click', closeCard);
    bar.append(x);
  } else bar.append(el('div', 'card-spacer'));
  return bar;
}

function renderLoading() {
  card.innerHTML = '';
  card.append(cardBar());
  const scroll = el('div', 'card-scroll');
  scroll.innerHTML = '<div style="text-align:center; padding:40px; color:var(--text-muted);">Завантаження...</div>';
  card.append(scroll);
}

async function loadMediaCardData(ctx) {
  const r = ctx.item;
  try {
    const d = await tmdbFetch(`/${r.type}/${r.id}?language=uk-UA&include_video_language=uk,en,null&append_to_response=credits,videos,external_ids,watch/providers,recommendations,similar,images`);
    ctx.details = d;
    
    if (!r.title) {
      r.title = d.title || d.name || '';
      r.year = (d.release_date || d.first_air_date || '').slice(0, 4);
      r.poster = d.poster_path;
      r.backdrop = d.backdrop_path;
      r.overview = d.overview || '';
      r.tmdbRating = d.vote_average || 0;
      r.votes = d.vote_count || 0;
      r.popularity = d.popularity || 0;
    }
    
    if (r.type === 'tv') r.episode_count = d.number_of_episodes || 0;
    if (d.genres) { ctx.genres = d.genres.map(g => g.id); if (lib[r.key]) { lib[r.key].genres = ctx.genres; save(); } }
    if (d.credits?.cast?.length) {
      ctx.cast = d.credits.cast.slice(0, CAST_LIMIT).map(a => ({ id: a.id, name: a.name, profile_path: a.profile_path }));
      if (lib[r.key] && !lib[r.key].cast) { lib[r.key].cast = ctx.cast; save(); }
    }
    if (d.external_ids?.imdb_id) {
      try {
        const omdbRes = await fetch(`https://www.omdbapi.com/?i=${d.external_ids.imdb_id}&apikey=${OMDB_KEY}`);
        const omdbData = await omdbRes.json();
        if (omdbData.Response === "True") ctx.omdb = omdbData;
      } catch {}
    }
    if (r.type === "tv" && d.seasons) {
      const sd = await Promise.all(d.seasons.map(s => tmdbFetch(`/tv/${r.id}/season/${s.season_number}?language=uk-UA`).catch(() => null)));
      ctx.details.seasons = d.seasons.map((s, i) => ({ ...s, episode_count: sd[i] ? sd[i].episodes.length : s.episode_count, episodes: sd[i] ? sd[i].episodes : [] }));
      const totalSeasons = d.seasons.filter(s => s.season_number !== 0).length;
      if (lib[r.key]) lib[r.key].totalSeasons = totalSeasons;
    }
    ctx.watchProviders = d['watch/providers']?.results?.UA || null;
    


    const recs = d.recommendations?.results || [];
    const sims = d.similar?.results || [];
    const pool = recs.length > 0 ? recs : sims;
    ctx.similar = pool.map(item => normalizeItem({ ...item, media_type: r.type })).filter(item => item.poster && item.id !== r.id);
  } catch (e) { ctx.error = e.message; }
}

function closeCard() {
  const iframe = card.querySelector('.card-hero-video');
  if (iframe) iframe.src = 'about:blank';

  if (cardStack.length) history.go(-cardStack.length);
  else { overlay.classList.remove('open'); refresh(); }
}

async function loadMediaForForward(s) {
  overlay.classList.add("open");
  const item = { key: s.key, id: s.id, type: s.media_type, title: '', year: '', poster: null, backdrop: null, overview: '', tmdbRating: 0, votes: 0, popularity: 0 };
  const ctx = { type: 'media', item, activeTab: 'cast', seasonState: {}, playingTrailer: false, overviewExpanded: false, logoPath: null };
  cardStack.push(ctx);
  renderLoading();
  await loadMediaCardData(ctx);
  if (cardStack[cardStack.length - 1] !== ctx) return;
  renderCard();
}

window.addEventListener('popstate', async (e) => {
  const depth = (e.state && e.state.mc) || 0;
  if (depth === cardStack.length) return;

  const iframe = card.querySelector('.card-hero-video');
  if (iframe) iframe.src = 'about:blank';

  if (depth > cardStack.length) {
    const s = e.state;
    if (!s || !s.ctx || depth - cardStack.length > 1) {
      const h = s?.ctx ? (s.ctx.type === 'person' ? '#/person/' + s.ctx.id : '#/' + s.ctx.media_type + '/' + s.ctx.id) : '';
      location.hash = h;
      location.reload();
      return;
    }
    if (s.ctx.type === 'person') await loadPersonForForward(s.ctx.id);
    else if (s.ctx.type === 'media') await loadMediaForForward(s.ctx);
    return;
  }
  while (cardStack.length > depth) cardStack.pop();
  if (depth === 0) { overlay.classList.remove('open'); refresh(); }
  else renderCard(true);
});

function routeFromHash() {
  const m = location.hash.match(/^#\/(movie|tv|person)\/(\d+)$/);
  if (!m) return;
  history.replaceState({}, '', location.pathname + location.search);
  const id = parseInt(m[2]);
  if (m[1] === 'person') openPerson(id, {});
  else openCard({ key: m[1] + '-' + id, id, type: m[1], title: '', year: '', poster: null, backdrop: null, overview: '', tmdbRating: 0, votes: 0, popularity: 0 });
}

function renderCard(restore) {
  const ctx = cardStack[cardStack.length - 1];
  if (!ctx) { overlay.classList.remove('open'); refresh(); return; }

  const oldIframe = card.querySelector('.card-hero-video');
  if (oldIframe) oldIframe.src = 'about:blank';

  const prevScroll = card.querySelector('.card-scroll');
  const savedScroll = restore ? (ctx.scroll || 0) : (prevScroll ? prevScroll.scrollTop : 0);
  card.innerHTML = '';
  card.append(cardBar());
  const scroll = el('div', 'card-scroll');
  card.append(scroll);
  if (!ctx.details && !ctx.error) scroll.innerHTML = '<div style="text-align:center; padding:40px; color:var(--text-muted);">Завантаження...</div>';
  else if (ctx.error) scroll.innerHTML = `<div style="text-align:center; padding:40px; color:var(--danger);">Помилка: ${ctx.error}</div>`;
  else if (ctx.type === 'person') renderPersonCard(ctx, scroll);
  else renderMediaCard(ctx, scroll);
  if (savedScroll > 0) scroll.scrollTop = savedScroll;
}

function renderNotesTab(ctx) {
  const r = ctx.item;
  const s = lib[r.key] || {};
  const wrap = el('div');
  wrap.append(el('div', 'notes-hint', 'Автоматично зберігаються'));
  const textarea = document.createElement('textarea');
  textarea.className = 'notes-textarea';
  textarea.placeholder = 'Свої думки, враження, цитати...';
  textarea.value = s.notes || '';
  let notesTimer;
  textarea.addEventListener('input', () => {
    clearTimeout(notesTimer);
    notesTimer = setTimeout(() => {
      const cur = lib[r.key] || { ...r, status: null, watchedEpisodes: [] };
      if (textarea.value.trim()) {
        lib[r.key] = { ...cur, notes: textarea.value, updated: Date.now() };
        if (!lib[r.key].createdAt) lib[r.key].createdAt = Date.now();
      } else if (lib[r.key]) {
        delete lib[r.key].notes;
        const hasR = lib[r.key].ratings && Object.values(lib[r.key].ratings).some(v => v > 0);
        const hasSR = lib[r.key].seasonRatings && Object.keys(lib[r.key].seasonRatings).length > 0;
        const hasEps = lib[r.key].watchedEpisodes && lib[r.key].watchedEpisodes.length > 0;
        if (!lib[r.key].status && !hasR && !hasSR && !hasEps) delete lib[r.key];
      }
      save();
    }, 800);
  });
  wrap.append(textarea);
  const status = el('div', 'notes-hint'); status.id = 'notes-status';
  if (s.notes) status.textContent = 'Оновлено: ' + new Date(s.updated || Date.now()).toLocaleString('uk-UA');
  wrap.append(status);
  return [wrap];
}

function toggleEpisodeWatched(r, seasonNum, epNum, ctx) {
  const s = lib[r.key] || {};
  let wEps = s.watchedEpisodes || [];
  if (s.status === 'watched' && ctx?.details?.seasons) {
    const all = [];
    ctx.details.seasons.forEach(season => {
      if (season.season_number === 0) return;
      const cnt = (season.episodes?.length) || season.episode_count || 0;
      for (let i = 1; i <= cnt; i++) all.push(season.season_number + '-' + i);
    });
    const wSet = new Set(wEps);
    all.forEach(k => wSet.add(k));
    wEps = [...wSet];
  }
  const epKey = seasonNum + '-' + epNum;
  if (wEps.includes(epKey)) wEps = wEps.filter(k => k !== epKey);
  else wEps = [...wEps, epKey];
  
  let totalAllEps = 0;
  if (ctx?.details?.seasons) {
    totalAllEps = ctx.details.seasons.filter(x => x.season_number !== 0).reduce((sum, x) => sum + (x.episodes?.length || x.episode_count || 0), 0);
  }
  let newStatus = s.status;
  if (wEps.length === 0) newStatus = s.status === 'watching' ? 'planned' : s.status;
  else if (totalAllEps > 0 && wEps.length >= totalAllEps) newStatus = 'watched';
  else newStatus = 'watching';
  
  const patch = { watchedEpisodes: wEps, status: newStatus };
  if (!lib[r.key] && ctx?.genres) patch.genres = ctx.genres;
  if (!lib[r.key] && ctx?.cast) patch.cast = ctx.cast;
  if (ctx?.details?.seasons && !s.totalSeasons) patch.totalSeasons = ctx.details.seasons.filter(x => x.season_number !== 0).length;
  update(r, patch);
}

function renderRatingBlock(crit, value, r) {
  const block = el("div", "rating-block");
  block.dataset.crit = crit.key;
  const header = el("div", "rating-block-header");
  header.append(el("span", "rating-block-label", crit.label));
  header.append(el("span", "rating-block-value" + (value ? "" : " empty"), value ? String(value) : "—"));
  block.append(header);
  const lights = el("div", "rating-lights");
  for (let i = 1; i <= 10; i++) {
    const light = el("span", "light" + (i <= value ? " on" : ""));
    light.addEventListener("click", () => setCriterion(r, crit.key, value === i ? 0 : i));
    lights.append(light);
  }
  block.append(lights);
  return block;
}

function renderAverageBlock(avg, filled) {
  const block = el("div", "rating-block average");
  const header = el("div", "rating-block-header");
  header.append(el("span", "rating-block-label", "Загальне (середнє)"));
  let text = avg > 0 ? fmtNum(avg) : "—";
  if (filled > 0 && filled < 5) text += ` (${filled}/5)`;
  header.append(el("span", "rating-block-value " + (avg > 0 ? ratingClass(avg) : 'empty'), text));
  block.append(header);
  const lights = el("div", "rating-lights");
  const full = Math.floor(avg);
  const frac = avg - full;
  for (let i = 1; i <= 10; i++) {
    const light = el("span", "light readonly");
    if (i <= full) light.classList.add("on");
    else if (i === full + 1 && frac > 0.01) {
      light.classList.add("partial");
      light.style.setProperty("--fill", (frac * 100).toFixed(0) + "%");
    }
    lights.append(light);
  }
  block.append(lights);
  return block;
}

function renderSeasonRatingBlock(r, seasonNum, crit, value) {
  const block = el("div", "rating-block");
  block.dataset.season = seasonNum;
  block.dataset.crit = crit.key;

function computeAverage(ratings) {
  if (!ratings) return 0;
  const vals = RATING_CRITERIA.map(c => ratings[c.key] || 0).filter(v => v > 0);
  return vals.length ? vals.reduce((a, b) => a + b, 0) / vals.length : 0;
}

function computeItemRating(item) {
  if (!item) return { avg: 0, complete: false, rated: 0, total: 0 };
  if (item.type === 'tv') {
    const sr = item.seasonRatings || {};
    const ratedKeys = Object.keys(sr).filter(k => computeAverage(sr[k]) > 0);
    const total = item.totalSeasons || ratedKeys.length || 0;
    if (!ratedKeys.length) return { avg: 0, complete: false, rated: 0, total };
    const avgs = ratedKeys.map(k => computeAverage(sr[k]));
    return { avg: avgs.reduce((a, b) => a + b, 0) / avgs.length, complete: total > 0 && ratedKeys.length >= total, rated: ratedKeys.length, total };
  }
  const avg = computeAverage(item.ratings);
  return { avg, complete: avg > 0, rated: avg > 0 ? 1 : 0, total: 1 };
}
function hasWatched(item) {
  if (!item) return false;
  if (item.status === 'watched' || item.status === 'watching') return true;
  if (item.ratings && Object.values(item.ratings).some(v => v > 0)) return true;
  if (item.seasonRatings && Object.keys(item.seasonRatings).length > 0) return true;
  if (item.watchedEpisodes && item.watchedEpisodes.length > 0) return true;
  return false;
}
function tasteDecay(updated) {
  const days = (Date.now() - (updated || Date.now())) / (24 * 60 * 60 * 1000);
  return Math.pow(0.98, days); // -3% щодня
}
function averagedSeasonRatings(item) {
  const sr = item.seasonRatings || {};
  const keys = Object.keys(sr).filter(k => Object.keys(sr[k] || {}).length > 0);
  if (!keys.length) return item.ratings || {};
  const sums = {}, counts = {};
  for (const k of keys) for (const crit of RATING_CRITERIA) {
    if (sr[k][crit.key]) { sums[crit.key] = (sums[crit.key] || 0) + sr[k][crit.key]; counts[crit.key] = (counts[crit.key] || 0) + 1; }
  }
  const r = {};
  for (const crit of RATING_CRITERIA) r[crit.key] = counts[crit.key] ? sums[crit.key] / counts[crit.key] : 0;
  return r;
}
function computeTasteVector() {
  const rated = Object.values(lib).map(x => ({ item: x, r: computeItemRating(x) })).filter(x => x.r.avg > 0);
  if (!rated.length) return null;
  const sums = {}, weights = {};
  for (const crit of RATING_CRITERIA) { sums[crit.key] = 0; weights[crit.key] = 0; }
  for (const { item } of rated) {
    const w = tasteDecay(item.updated);
    const rr = item.type === 'tv' ? averagedSeasonRatings(item) : (item.ratings || {});
    for (const crit of RATING_CRITERIA) if (rr[crit.key]) { sums[crit.key] += rr[crit.key] * w; weights[crit.key] += w; }
  }
  const result = { count: rated.length, movieCount: rated.filter(x => x.item.type === 'movie').length, tvCount: rated.filter(x => x.item.type === 'tv').length };
  for (const crit of RATING_CRITERIA) result[crit.key] = weights[crit.key] > 0 ? sums[crit.key] / weights[crit.key] : 0;
  return result;
}

async function startGame(type) {
  state.game = { active: true, type, round: 0, totalRounds: 20, champion: null, challenger: null, history: [], finished: false, pool: [], seen: new Set(), usedPages: new Set() };
  renderGame();
  await loadGameChampion();
  await loadGameChallenger();
  renderGame();
}

async function refillGamePool() {
  const g = state.game;
  const path = g.type === 'movie' ? '/discover/movie' : '/discover/tv';
  for (let a = 0; a < 6 && g.pool.length < 6; a++) {
    let page;
    do { page = Math.floor(Math.random() * 15) + 1; } while (g.usedPages.has(page) && g.usedPages.size < 15);
    g.usedPages.add(page);
    try {
      const d = await tmdbFetch(`${path}?language=uk-UA&sort_by=popularity.desc&vote_count.gte=300&include_adult=false&page=${page}`);
      d.results.map(r => normalizeItem({ ...r, media_type: g.type }))
        .filter(r => r.poster && !g.seen.has(r.key) && !g.pool.some(x => x.key === r.key))
        .forEach(r => g.pool.push(r));
    } catch (e) { console.error(e); }
  }
}

function takeFromPool() {
  const g = state.game;
  if (!g.pool.length) return null;
  const item = g.pool.splice(Math.floor(Math.random() * g.pool.length), 1)[0];
  g.seen.add(item.key);
  return item;
}

async function loadGameChampion() { state.game.pool = []; await refillGamePool(); state.game.champion = takeFromPool(); }
async function loadGameChallenger() { if (state.game.pool.length < 3) await refillGamePool(); state.game.challenger = takeFromPool(); }

function makeGameChoice(choice) {
  if (!state.game.champion || !state.game.challenger) return;
  let winner, result, eloDelta = 0;
  if (choice === 'challenger') {
    winner = state.game.challenger;
    const eloResult = recordEloMatch(state.game.challenger, state.game.champion);
    if (eloResult) eloDelta = eloResult.delta;
    state.game.champion = winner;
    result = 'challenger';
  } else if (choice === 'champion') {
    winner = state.game.champion;
    const eloResult = recordEloMatch(state.game.champion, state.game.challenger);
    if (eloResult) eloDelta = eloResult.delta;
    result = 'champion';
  } else {
    winner = state.game.champion;
    result = 'skip';
  }
  state.game.history.push({ round: state.game.round, winner, result, eloDelta, champion: state.game.champion, challenger: state.game.challenger });
  state.game.round++;
  if (state.game.round >= state.game.totalRounds) { finishGame(); return; }
  loadGameChallenger().then(() => renderGame());
}

function finishGame() {
  state.game.finished = true;
  state.game.active = false;
  const hist = JSON.parse(localStorage.getItem('kingOfTheHillHistory') || '[]');
  hist.push({ date: new Date().toISOString(), wins: state.game.history.filter(h => h.result !== 'skip').length, total: state.game.totalRounds, type: state.game.type, champion: state.game.champion });
  localStorage.setItem('kingOfTheHillHistory', JSON.stringify(hist));
  renderGame();
}

  const c = el('div', 'game-container game-over');
  c.append(el('h2', '', '🎉 Гра завершена!'));
  c.append(el('p', '', 'Твій фінальний чемпіон:'));

    cd.append(p);
    const body = el('div', 'game-card-body');
    body.append(el('div', 'game-card-title', state.game.champion.title));
    body.append(el('div', 'game-card-elo', `Elo ${Math.round(getElo(state.game.champion))}`));

// ============ RECOMMENDATIONS LOGIC ============
async function buildSmartRecommendations(seeds, mood) {
  if (!seeds.length) return { items: [], seeds: 0, mood };

  const seedDetails = await Promise.all(seeds.slice(0, 6).map(async s => {
    try { const d = await tmdbFetch(`/${s.type}/${s.id}?language=uk-UA&append_to_response=credits,keywords`); return { seed: s, data: d }; } catch { return null; }
  })).then(arr => arr.filter(Boolean));
  
  const signals = { directors: {}, cast: {}, genres: {}, keywords: {} };
  
  // Оновлені ваги: Режисер 2.5, Актор 2.2, Жанр 1.5, Ключові слова 1.2
  for (const { seed, data } of seedDetails) {
    const w = (seed.avg / 10) * tasteDecay(seed.updated); // tasteDecay тепер 3% на день
    (data.credits?.crew || []).filter(c => c.job === 'Director' || c.job === 'Creator').forEach(c => {
      signals.directors[c.id] = (signals.directors[c.id] || 0) + w * 2.5;
    });
    (data.credits?.cast || []).slice(0, 5).forEach(c => {
      signals.cast[c.id] = (signals.cast[c.id] || 0) + w * 2.2;
    });
    (data.genres || []).forEach(g => {
      signals.genres[g.id] = (signals.genres[g.id] || 0) + w * 1.5;
    });
    const kws = data.keywords?.keywords || data.keywords?.results || [];
    kws.slice(0, 12).forEach(k => {
      signals.keywords[k.id] = (signals.keywords[k.id] || 0) + w * 1.2;
    });
  }
  
  const moodData = mood && MOODS[mood] ? MOODS[mood] : null;
  const topKw = Object.entries(signals.keywords).sort((a, b) => b[1] - a[1]).slice(0, 5).map(([id]) => id);
  const topGenres = Object.entries(signals.genres).sort((a, b) => b[1] - a[1]).slice(0, 4).map(([id]) => parseInt(id));
  const topDirectors = Object.entries(signals.directors).sort((a, b) => b[1] - a[1]).slice(0, 2).map(([id]) => id);
  
  const fetchPromises = [];
  const pages = [1, 2, 3, 4]; // Тягнемо 4 сторінки, щоб назбирати більше унікальних тайтлів

  // MOVIES
  {
    const p = new URLSearchParams({ language: 'uk-UA', sort_by: 'popularity.desc', 'vote_count.gte': '50', 'vote_average.gte': '5.5' });
    if (topKw.length) p.set('with_keywords', topKw.join('|'));
    if (topGenres.length) p.set('with_genres', topGenres.slice(0, 3).join('|'));
    pages.forEach(page => {
      p.set('page', page);
      fetchPromises.push(tmdbFetch(`/discover/movie?${p}`).then(d => ({ type: 'movie', results: d.results || [], source: 'personalized' })).catch(() => ({ type: 'movie', results: [], source: 'personalized' })));
    });

    if (moodData) {
      const pm = new URLSearchParams({ language: 'uk-UA', sort_by: 'popularity.desc', 'vote_count.gte': '80', 'vote_average.gte': '6', with_genres: moodData.genres.join('|') });
      pages.forEach(page => {
        pm.set('page', page);
        fetchPromises.push(tmdbFetch(`/discover/movie?${pm}`).then(d => ({ type: 'movie', results: d.results || [], source: 'mood' })).catch(() => ({ type: 'movie', results: [], source: 'mood' })));
      });
    }
  }
  
  // TV
  {
    const p = new URLSearchParams({ language: 'uk-UA', sort_by: 'popularity.desc', 'vote_count.gte': '50', 'vote_average.gte': '5.5' });
    if (topKw.length) p.set('with_keywords', topKw.join('|'));
    pages.forEach(page => {
      p.set('page', page);
      fetchPromises.push(tmdbFetch(`/discover/tv?${p}`).then(d => ({ type: 'tv', results: d.results || [], source: 'personalized' })).catch(() => ({ type: 'tv', results: [], source: 'personalized' })));
    });

    if (topGenres.length) {
      const pg = new URLSearchParams({ language: 'uk-UA', sort_by: 'popularity.desc', 'vote_count.gte': '80', 'vote_average.gte': '6', with_genres: topGenres.slice(0, 3).join('|') });
      pages.forEach(page => {
        pg.set('page', page);
        fetchPromises.push(tmdbFetch(`/discover/tv?${pg}`).then(d => ({ type: 'tv', results: d.results || [], source: 'genres' })).catch(() => ({ type: 'tv', results: [], source: 'genres' })));
      });
    }

    if (moodData) {
      const pm = new URLSearchParams({ language: 'uk-UA', sort_by: 'popularity.desc', 'vote_count.gte': '80', 'vote_average.gte': '6', with_genres: (moodData.tvGenres || moodData.genres).join('|') });
      pages.forEach(page => {
        pm.set('page', page);
        fetchPromises.push(tmdbFetch(`/discover/tv?${pm}`).then(d => ({ type: 'tv', results: d.results || [], source: 'mood' })).catch(() => ({ type: 'tv', results: [], source: 'mood' })));
      });
    }
  }
  
  topDirectors.forEach(dId => {
    fetchPromises.push(tmdbFetch(`/discover/movie?language=uk-UA&with_crew=${dId}&sort_by=popularity.desc`).then(d => ({ type: 'movie', results: d.results || [], source: 'director' })).catch(() => ({ type: 'movie', results: [], source: 'director' })));
  });
  
  const all = await Promise.all(fetchPromises);
  const unique = new Map();
  
  for (const { type, results, source } of all) {
    for (const r of results) {
      if (!r.poster_path || lib[`${type}-${r.id}`]) continue;
      const item = normalizeItem({ ...r, media_type: type });
      let entry = unique.get(item.key);
      if (!entry) { entry = { ...item, _score: 0, _reasons: new Set() }; unique.set(item.key, entry); }
      
      let score = (item.tmdbRating / 10) + Math.min(item.popularity / 100, 1);
      if (source === 'mood') { score += 2.0; entry._reasons.add(moodData.emoji + ' ' + moodData.label); }
      if (source === 'director') { score += 1.5; entry._reasons.add('той самий режисер'); }
      if (source === 'personalized') score += 0.8;
      if (source === 'genres') score += 0.5;
      
      entry._score += score;
      
      const matchedGenres = (item.genres || []).filter(g => topGenres.includes(g));
      if (matchedGenres.length && entry._reasons.size < 2) {
        const g = GENRES.find(x => matchedGenres.includes(x.id));
        if (g) entry._reasons.add(`жанр «${g.name}»`);
      }
    }
  }
  
  const byType = { movie: [], tv: [] };
  for (const item of unique.values()) { 
    item._reasons = [...item._reasons].slice(0, 2); 
    byType[item.type].push(item); 
  }
  
  // Беремо топ-200, а потім перемішуємо (Fisher-Yates)
  const pickAndShuffle = (t) => {
    const sorted = byType[t].sort((a, b) => b._score - a._score).slice(0, 200);
    for (let i = sorted.length - 1; i > 0; i--) {
      const j = Math.floor(Math.random() * (i + 1));
      [sorted[i], sorted[j]] = [sorted[j], sorted[i]];
    }
    return sorted;
  };

  return { items: [...pickAndShuffle('movie'), ...pickAndShuffle('tv')], seeds: seeds.length, mood };
}

const RECS_CACHE = 'mycinema.recs.v2';
const RECS_TTL = 6 * 60 * 60 * 1000;

function getSeeds() {
  return Object.values(lib).map(x => ({ ...x, avg: computeItemRating(x).avg }))
    .filter(x => x.avg >= 7 && x.id)
    .sort((a, b) => (b.updated || 0) - (a.updated || 0)).slice(0, 10);
}
function readRecsCache(sig) {
  try { const c = JSON.parse(localStorage.getItem(RECS_CACHE)); if (c && c.sig === sig && Date.now() - c.ts < RECS_TTL) return c.data; } catch {}
  return null;
}
function writeRecsCache(sig, data) { try { localStorage.setItem(RECS_CACHE, JSON.stringify({ sig, ts: Date.now(), data })); } catch {} }

// ============ TASTE DNA (SVG & UI) ============
function tasteRadarSVG(taste, size = 240) {
  const cx = size / 2, cy = size / 2, r = size / 2 - 42;
  const n = RATING_CRITERIA.length;
  const pts = RATING_CRITERIA.map((c, i) => {
    const angle = -Math.PI / 2 + (i * 2 * Math.PI / n);
    const v = (taste[c.key] || 0) / 10;
    return { x: cx + r * v * Math.cos(angle), y: cy + r * v * Math.sin(angle), ax: cx + r * Math.cos(angle), ay: cy + r * Math.sin(angle), lx: cx + (r + 22) * Math.cos(angle), ly: cy + (r + 22) * Math.sin(angle), label: c.label.split(' ')[0] };
  });
  const grid = [1, 2, 3, 4].map(level => {
    const lr = r * level / 4;
    const gpts = RATING_CRITERIA.map((c, i) => { const a = -Math.PI / 2 + i * 2 * Math.PI / n; return `${cx + lr * Math.cos(a)},${cy + lr * Math.sin(a)}`; }).join(' ');
    return `<polygon points="${gpts}" fill="none" stroke="rgba(255,255,255,0.05)" stroke-width="1"/>`;
  }).join('');
  const axes = pts.map(p => `<line x1="${cx}" y1="${cy}" x2="${p.ax}" y2="${p.ay}" stroke="rgba(255,255,255,0.04)" stroke-width="1"/>`).join('');
  const labels = pts.map(p => `<text x="${p.lx}" y="${p.ly}" text-anchor="middle" dominant-baseline="middle" fill="var(--text-muted)" font-size="9.5" font-family="system-ui">${p.label}</text>`).join('');
  const poly = pts.map(p => `${p.x},${p.y}`).join(' ');
  const dots = pts.map(p => `<circle cx="${p.x}" cy="${p.y}" r="3.5" fill="var(--accent)"/>`).join('');
  return `<svg viewBox="0 0 ${size} ${size}" style="width:100%;max-width:${size}px;display:block">${grid}${axes}<polygon points="${poly}" fill="rgba(var(--accent-rgb),0.18)" stroke="var(--accent)" stroke-width="2" stroke-linejoin="round"/>${dots}${labels}</svg>`;
}

function renderTasteInsight(taste) {
  if (!taste || taste.count < 3) return null;
  const vals = RATING_CRITERIA.map(c => ({ key: c.key, label: c.label, v: taste[c.key] }));
  const sorted = [...vals].sort((a, b) => b.v - a.v);
  const top = sorted[0], bottom = sorted[sorted.length - 1];
  const gap = top.v - bottom.v;
  let desc;
  if (gap < 0.8) desc = `Ти — <em>збалансований глядач</em>. Усі критерії для тебе однаково важливі.`;
  else if (top.key === 'plot') desc = `Ти — <em>наративний глядач</em>. Історія та сюжет для тебе важливіші за форму.`;
  else if (top.key === 'cast') desc = `Ти — <em>персонажний глядач</em>. Ти цінуєш акторів і хімію між ними.`;
  else if (top.key === 'style') desc = `Ти — <em>візуальний глядач</em>. Картинка, атмосфера й стиль для тебе критичні.`;
  else if (top.key === 'emotions') desc = `Ти — <em>емоційний глядач</em>. Головне — що ти відчуваєш під час перегляду.`;
  else if (top.key === 'overall') desc = `Ти — <em>інтуїтивний глядач</em>. Довіряєш загальному враженню більше за деталі.`;
  else desc = `Твій найсильніший критерій — <em>${top.label}</em>.`;
  return { top, bottom, desc };
}

  }
  body.append(rows);
  card.append(body);

  const insight = renderTasteInsight(taste);
  if (insight) {
    const ins = el('div', 'taste-insight');
    ins.innerHTML = insight.desc;
    card.append(ins);
  }
  return card;
}

function renderTasteSummary(taste) {
  const container = el('div', 'taste-summary-container');
  const hasTaste = taste && taste.count >= 3;

  let summaryVal = '—';
  if (hasTaste) {
    const vals = RATING_CRITERIA.map(c => taste[c.key] || 0).filter(v => v > 0);
    if (vals.length) summaryVal = fmtNum(vals.reduce((a, b) => a + b, 0) / vals.length) + '/10';
  }

  const btn = el('button', 'taste-summary');
  btn.append(el('span', 'taste-summary-icon', '🧬'));
  btn.append(el('span', 'taste-summary-label', 'Твій смак у цифрах'));
  btn.append(el('span', 'taste-summary-value', summaryVal));
  const chev = el('span', 'taste-summary-chevron', '▾');
  btn.append(chev);
  if (state.home.tasteExpanded) btn.classList.add('expanded');

  const panel = el('div', 'taste-panel');
  panel.style.display = state.home.tasteExpanded ? 'block' : 'none';

  if (hasTaste) {
    panel.append(buildTasteCard(taste));
  } else {
    const empty = el('div', 'taste-card');
    empty.append(el('div', 'taste-empty', 'Оціни хоча б 3 тайтли — і тут з\'явиться радарна карта твого смаку.'));

  }

    btn.classList.toggle('expanded', state.home.tasteExpanded);
    panel.style.display = state.home.tasteExpanded ? 'block' : 'none';
  });

  container.append(btn, panel);
  return container;
}

// ============ LOAD & RENDER ============
async function loadHome() {
  if (state.tab !== 'home') return;
  const seeds = getSeeds();
  const taste = computeTasteVector();
  const sig = (seeds.map(s => s.key + ':' + s.avg.toFixed(1)).join('|')) + '|mood:' + (state.home.mood || 'any');

  if (!state.home.general) {
    try {
      const [trM, newM, topM, trT, newT, topT] = await Promise.all([
        tmdbFetch('/trending/movie/week?language=uk-UA').catch(() => ({ results: [] })),
        tmdbFetch('/discover/movie?language=uk-UA&sort_by=primary_release_date.desc&vote_count.gte=50&page=1').catch(() => ({ results: [] })),
        tmdbFetch('/discover/movie?language=uk-UA&sort_by=vote_average.desc&vote_count.gte=1000&page=1').catch(() => ({ results: [] })),
        tmdbFetch('/trending/tv/week?language=uk-UA').catch(() => ({ results: [] })),
        tmdbFetch('/discover/tv?language=uk-UA&sort_by=first_air_date.desc&vote_count.gte=50&page=1').catch(() => ({ results: [] })),
        tmdbFetch('/discover/tv?language=uk-UA&sort_by=vote_average.desc&vote_count.gte=500&page=1').catch(() => ({ results: [] }))
      ]);
      state.home.general = {
        trMovies: trM.results.map(r => normalizeItem({ ...r, media_type: 'movie' })).filter(x => x.poster).slice(0, 10),
        newMovies: newM.results.map(r => normalizeItem({ ...r, media_type: 'movie' })).filter(x => x.poster).slice(0, 10),
        topMovies: topM.results.map(r => normalizeItem({ ...r, media_type: 'movie' })).filter(x => x.poster).slice(0, 10),
        trTV: trT.results.map(r => normalizeItem({ ...r, media_type: 'tv' })).filter(x => x.poster).slice(0, 10),
        newTV: newT.results.map(r => normalizeItem({ ...r, media_type: 'tv' })).filter(x => x.poster).slice(0, 10),
        topTV: topT.results.map(r => normalizeItem({ ...r, media_type: 'tv' })).filter(x => x.poster).slice(0, 10)
      };
    } catch (e) { console.error(e); }
  }

  if (!seeds.length) {
    state.home.recs = { items: [], seeds: 0, mood: state.home.mood };
    state.home.sig = sig;
    renderHomeContent(taste);
    return;
  }
  if (state.home.recs && state.home.sig === sig) { renderHomeContent(taste); return; }
  const cached = readRecsCache(sig);
  if (cached) { state.home.recs = cached; state.home.sig = sig; renderHomeContent(taste); return; }
  if (state.home.loading === sig) return;

  state.home.loading = sig;
  box.style.padding = '0';
  box.innerHTML = '<div style="text-align:center; padding:40px; color:var(--text-muted);">Підбираю рекомендації...</div>';
  try {
    const data = await buildSmartRecommendations(seeds, state.home.mood);
    writeRecsCache(sig, data);
    state.home.recs = data;
    state.home.sig = sig;
    state.home.shown = { movie: 20, tv: 20 }; // Скидаємо лічильник при новому завантаженні
    renderHomeContent(taste);
  } catch (e) { box.innerHTML = `<div style="text-align:center; padding:40px; color:var(--danger);">Помилка: ${e.message}</div>`; }
  finally { state.home.loading = false; }
}

// ========== LIBRARY LIST ==========
function sortLibraryItems(items) {
  const s = state.librarySort;
  return items.slice().sort((a, b) => {
    if (s === 'rating') {
      const aR = computeItemRating(a).avg;
      const bR = computeItemRating(b).avg;
      return bR - aR || (b.updated || 0) - (a.updated || 0);
    }
    if (s === 'title') {
      return (a.title || '').localeCompare(b.title || '', 'uk');
    }
    if (s === 'created') {
      return (b.createdAt || b.updated || 0) - (a.createdAt || a.updated || 0);
    }
    return (b.updated || 0) - (a.updated || 0);
  });
}

function createListItem(r, showNumber, number) {
  const item = el("div", "item" + (showNumber ? " has-number" : ""));
  if (showNumber) item.append(el("div", "top-number", String(number)));
  item.append(posterEl(r, "poster"));
  const info = el("div", "info");
  info.append(el("div", "title", r.title));
  
  let metaText = (r.type === "tv" ? "Серіал" : "Фільм") + (r.year ? " · " + r.year : "");
  if (r.tmdbRating > 0) metaText += ` · ${r.tmdbRating.toFixed(1)}`;
  info.append(el("div", "meta", metaText));
  
  const saved = lib[r.key];
  if (saved) {
    const ir = computeItemRating(saved);
    if (ir.avg > 0) {
      const suffix = ir.complete ? '' : '-...';
      info.append(el("span", "badge rated " + ratingClass(ir.avg), `${fmtNum(ir.avg)}${suffix}`));
    } else if (saved.status) {
      const labels = { watched: "✅", planned: "📅", watching: "▶️", dropped: "🕸️" };
      info.append(el("span", "badge", labels[saved.status] || ""));
    }
    if (saved.elo && saved.eloGames > 1) {
      info.append(el("span", "badge elo", `🏆 ${saved.elo}`));
    }
    if (r.type === 'tv' && saved.watchedEpisodes && saved.watchedEpisodes.length > 0 && saved.totalSeasons) {
      const total = saved.watchedEpisodes.length;
      info.append(el('div', 'meta', `${total} ${plural(total, 'серія', 'серії', 'серій')}`));
    }
  }
  item.append(info);
  
  if (saved) {
    const acts = el('div', 'item-actions');
    const favBtn = el('button', 'item-fav' + (saved.favorite ? ' on' : ''), '❤');
    favBtn.title = 'В улюблене';
    favBtn.addEventListener('click', (e) => { e.stopPropagation(); update(r, { favorite: !saved.favorite }); });
    acts.append(favBtn);
    const del = el("button", "item-delete", "×");
    del.title = "Прибрати зі списку";
    del.addEventListener("click", (e) => { e.stopPropagation(); removeFromLibrary(r.key); });
    acts.append(del);
    item.append(acts);
  }
  item.addEventListener("click", () => openCard(r));
  return item;
}

function removeFromLibrary(key) {
  if (!lib[key]) return;
  if (!confirm('Прибрати цей тайтл зі списку?\nОцінки та нотатки збережуться.')) return;
  const saved = lib[key];
  const hasR = saved.ratings && Object.values(saved.ratings).some(v => v > 0);
  const hasSR = saved.seasonRatings && Object.keys(saved.seasonRatings).length > 0;
  const hasNotes = saved.notes && saved.notes.trim();
  const hasElo = saved.eloGames && saved.eloGames > 0;
  if (hasR || hasSR || hasNotes || hasElo) {
    lib[key].status = null;
    lib[key].updated = Date.now();
    saveNow();
  } else {
    delete lib[key];
    saveNow();
  }
  refresh();
  const ctx = cardStack[cardStack.length - 1];
  if (ctx && ctx.type === 'media' && ctx.item.key === key) renderCard();
  showSyncToast('🗑️ Прибрано');
}

function renderFilter() {
  if (box.querySelector('.filter-panel')) {
    filterPage = 1;
    const list = document.getElementById('filter-results-list'); if (list) list.innerHTML = '';
    doFetchFilter(false); return;
  }
  box.replaceChildren(); box.style.padding = '0 16px 16px';
  if (state.filter.query) {
    const qb = el("div", "query-bar");
    qb.append(el("span", "", `Результати пошуку: "${state.filter.query}"`));
    const clr = el("button", "", "✕");
    clr.addEventListener("click", () => { state.filter.query = ''; input.value = ''; renderFilter(); });
    qb.append(clr); box.append(qb);
  }
  box.append(buildFilterPanel());
  const ra = el("div"); ra.id = "filter-results-list"; box.append(ra);
  filterPage = 1; doFetchFilter(false);
}

function buildFilterPanel() {
  const panel = el("div", "filter-panel");
  const decades = [['any', 'Будь-яке'], ['2020', '2020-ті'], ['2010', '2010-ті'], ['2000', '2000-ті'], ['1990', '1990-ті'], ['1980', '1980-ті'], ['1970', '1970-ті']];
  const ratings = [[0, 'Будь-який'], [5, '5+'], [6, '6+'], [7, '7+'], [8, '8+'], [9, '9+']];
  const sorts = [['popularity.desc', 'Популярність ↓'], ['vote_average.desc', 'Рейтинг ↓'], ['primary_release_date.desc', 'Новіші'], ['primary_release_date.asc', 'Старіші']];
  const streams = [['any', 'Будь-який'], ...STREAMING_PROVIDERS.map(p => [String(p.id), p.name])];
  
  panel.append(makeSelectRow('Тип', 'f-type', [['all', 'Всі'], ['movie', 'Фільми'], ['tv', 'Серіали']], state.filter.type));
  panel.append(makeSelectRow('Десятиліття', 'f-decade', decades, state.filter.decade));
  panel.append(makeSelectRow('Жанр', 'f-genre', [['any', 'Будь-який'], ...GENRES.map(g => [String(g.id), g.name])], String(state.filter.genre)));
  panel.append(makeSelectRow('Мін. рейтинг', 'f-rating', ratings, String(state.filter.minRating)));
  panel.append(makeSelectRow('Стрімінг', 'f-streaming', streams, state.filter.streaming ? String(state.filter.streaming) : 'any'));
  panel.append(makeSelectRow('Сортування', 'f-sort', sorts, state.filter.sortBy));
  
  const actions = el("div", "filter-actions");
  const resetBtn = el("button", "", "Скинути");
  resetBtn.addEventListener("click", () => {
    state.filter = { ...state.filter, type: 'all', decade: 'any', genre: 'any', minRating: 0, sortBy: 'popularity.desc', streaming: null };
    ['f-type','f-decade','f-genre','f-rating','f-sort','f-streaming'].forEach(id => document.getElementById(id).value = id === 'f-type' ? 'all' : id === 'f-decade' ? 'any' : id === 'f-genre' ? 'any' : id === 'f-rating' ? '0' : id === 'f-sort' ? 'popularity.desc' : 'any');
    filterPage = 1;
    const list = document.getElementById('filter-results-list'); if (list) list.innerHTML = '';
    doFetchFilter(false);
  });
  actions.append(resetBtn); panel.append(actions);
  return panel;
}

function makeSelectRow(label, id, options, currentVal) {
  const row = el("div", "filter-row");
  row.append(el("label", "", label));
  const sel = document.createElement("select"); sel.id = id;
  for (const [val, text] of options) {
    const opt = document.createElement("option"); opt.value = val; opt.textContent = text;
    if (String(val) === String(currentVal)) opt.selected = true;
    sel.append(opt);
  }
  sel.addEventListener("change", () => {
    if (id === 'f-type') state.filter.type = sel.value;
    else if (id === 'f-decade') state.filter.decade = sel.value;
    else if (id === 'f-genre') state.filter.genre = sel.value === 'any' ? 'any' : parseInt(sel.value);
    else if (id === 'f-rating') state.filter.minRating = parseInt(sel.value);
    else if (id === 'f-sort') state.filter.sortBy = sel.value;
    else if (id === 'f-streaming') state.filter.streaming = sel.value === 'any' ? null : parseInt(sel.value);
    filterPage = 1;
    const list = document.getElementById('filter-results-list'); if (list) list.innerHTML = '';
    doFetchFilter(false);
  });
  row.append(sel);
  return row;
}

async function doFetchFilter(append) {
  if (filterLoading) return;
  filterLoading = true;
  const resultsList = document.getElementById('filter-results-list');
  if (!resultsList) { filterLoading = false; return; }
  if (!append) resultsList.innerHTML = '<div style="text-align:center; padding:20px; color:var(--text-muted);">Пошук...</div>';
  const f = state.filter;
  
  try {
    let hasMore = false;
    const people = [], media = [];
    if (f.query) {
      const data = await tmdbFetch(`/search/multi?language=uk-UA&query=${encodeURIComponent(f.query)}&include_adult=false&page=${filterPage}`);
      if (!append) data.results.filter(r => r.media_type === 'person').forEach(p => people.push(p));
      data.results.filter(r => r.media_type === 'movie' || r.media_type === 'tv').map(normalizeItem).forEach(x => media.push(x));
      hasMore = filterPage < (data.total_pages || 1);
    } else {
      const bp = { language: 'uk-UA', sort_by: (f.sortBy.includes('release_date') && f.type === 'all') ? 'popularity.desc' : f.sortBy, page: String(filterPage) };
      if (f.minRating > 0) bp['vote_average.gte'] = String(f.minRating);
      if (f.genre !== "any") bp['with_genres'] = String(f.genre);
      if (f.streaming) { bp['with_watch_providers'] = String(f.streaming); bp['watch_region'] = 'UA'; }
      
      const fetches = [];
      let mP = 0, tP = 0;
      if (f.type === "movie" || f.type === "all") {
        const p = { ...bp };
        if (f.decade !== "any") { p['primary_release_date.gte'] = `${f.decade}-01-01`; p['primary_release_date.lte'] = `${parseInt(f.decade) + 9}-12-31`; }
        fetches.push(tmdbFetch(`/discover/movie?${new URLSearchParams(p)}`).then(d => { mP = d.total_pages || 1; return d.results.map(r => normalizeItem({ ...r, media_type: 'movie' })); }).catch(() => []));
      }
      if (f.type === "tv" || f.type === "all") {
        const p = { ...bp };
        if (f.decade !== "any") { p['first_air_date.gte'] = `${f.decade}-01-01`; p['first_air_date.lte'] = `${parseInt(f.decade) + 9}-12-31`; }
        fetches.push(tmdbFetch(`/discover/tv?${new URLSearchParams(p)}`).then(d => { tP = d.total_pages || 1; return d.results.map(r => normalizeItem({ ...r, media_type: 'tv' })); }).catch(() => []));
      }
      const all = await Promise.all(fetches);
      all.flat().forEach(x => media.push(x));
      hasMore = filterPage < Math.max(mP, tP);
      if (f.type === "all") {
        if (f.sortBy === 'popularity.desc') media.sort((a, b) => b.popularity - a.popularity);
        else if (f.sortBy === 'vote_average.desc') media.sort((a, b) => b.tmdbRating - a.tmdbRating);
      }
    }
    
    if (!append) resultsList.innerHTML = '';
    if (f.minRating > 0) for (let i = media.length - 1; i >= 0; i--) if (media[i].tmdbRating < f.minRating) media.splice(i, 1);
    if (f.decade !== "any") {
      const yS = parseInt(f.decade), yE = yS + 9;
      for (let i = media.length - 1; i >= 0; i--) { const y = parseInt(media[i].year); if (!(y >= yS && y <= yE)) media.splice(i, 1); }
    }
    
    for (const p of people) resultsList.append(createPersonResult(p));
    for (const r of media) resultsList.append(createListItem(r, false));
    
    const old = document.getElementById('filter-more-btn'); if (old) old.remove();
    if (hasMore) {
      const more = el('button', 'home-more', 'Показати ще'); more.id = 'filter-more-btn';
      more.addEventListener('click', () => { filterPage++; doFetchFilter(true); });
      resultsList.append(more);
    }
    if (!resultsList.children.length) resultsList.innerHTML = '<div style="text-align:center; padding:20px; color:var(--text-muted);">Нічого не знайдено</div>';
  } catch (e) {
    resultsList.innerHTML = `<div style="text-align:center; padding:20px; color:var(--danger);">Помилка: ${e.message}</div>`;
  } finally { filterLoading = false; }
}

async function openPerson(personId, fromItem) {
  overlay.classList.add("open");
  const top = cardStack[cardStack.length - 1];
  if (top && top.type === 'person' && top.id === personId) return;
  saveCardScroll();
  const ctx = { type: 'person', id: personId, item: fromItem, personTab: 'bio' };
  cardStack.push(ctx);
  pushCardState();
  renderLoading();
  await loadPersonCardData(ctx);
  if (cardStack[cardStack.length - 1] !== ctx) return;
  renderCard();
}

async function loadPersonCardData(ctx) {
  try {
    const [details, imagesData] = await Promise.all([
      tmdbFetch(`/person/${ctx.id}?language=uk-UA&append_to_response=combined_credits`),
      tmdbFetch(`/person/${ctx.id}/images`).catch(() => ({ profiles: [] }))
    ]);
    ctx.details = details;
    ctx.images = (imagesData?.profiles || []).filter(p => p.file_path);
  } catch (e) { ctx.error = e.message; }
}

async function loadPersonForForward(id) {

  cardStack.push(ctx);
  renderLoading();

}

function renderPersonCard(ctx, scroll) {
  const d = ctx.details;
  if (!d) return;
  const header = el('div', 'person-header');
  const photo = el(d.profile_path ? 'img' : 'div', 'person-photo');
  if (d.profile_path) {
    photo.src = TMDB_IMG + 'w342' + d.profile_path;
    photo.classList.add('zoomable');
    photo.addEventListener('click', () => openImageZoom(d.profile_path, 'w500'));
  }
  header.append(photo);
  const info = el('div');
  info.append(el('div', 'person-name', d.name || ''));
  if (d.known_for_department) info.append(el('div', 'person-meta', d.known_for_department));
  if (d.birthday) {
    let b = d.birthday;
    if (d.deathday) b += ' — ' + d.deathday;
    info.append(el('div', 'person-meta', 'Народження: ' + b));
  }
  if (d.place_of_birth) info.append(el('div', 'person-meta', d.place_of_birth));
  header.append(info);
  scroll.append(header);
  
  if (!ctx.personTab) ctx.personTab = 'bio';
  const tabs = el('div', 'card-tabs');
  const tabDefs = [
    { id: 'bio', label: '📖 Біографія' },
    { id: 'filmography', label: '🎬 Фільмографія' },
    { id: 'gallery', label: '📸 Галерея' }
  ];
  for (const t of tabDefs) {
    const btn = el('button', 'card-tab' + (ctx.personTab === t.id ? ' active' : ''), t.label);
    btn.addEventListener('click', () => { if (ctx.personTab === t.id) return; ctx.personTab = t.id; renderCard(); });
    tabs.append(btn);
  }
  scroll.append(tabs);
  
  const content = el('div'); content.style.marginTop = '16px';
  if (ctx.personTab === 'bio') {
    if (d.biography) content.append(el('div', 'person-bio', d.biography));
    else content.append(el('div', 'avail-empty', 'Біографія відсутня'));
  } else if (ctx.personTab === 'filmography') {
    content.append(...renderPersonFilmography(ctx));
  } else if (ctx.personTab === 'gallery') {
    content.append(...renderPersonGallery(ctx));
  }
  scroll.append(content);
}

function renderPersonFilmography(ctx) {
  const d = ctx.details;
  const allCast = (d.combined_credits?.cast || []).filter(c => c.poster_path);
  const filtered = allCast.filter(c => {
    const genres = c.genre_ids || [];
    if (genres.includes(ANIMATION_GENRE_ID)) return false;
    if (c.media_type === 'tv' && genres.some(g => EXCLUDED_TV_GENRES.includes(g))) return false;
    return true;
  });
  
  const yearOf = c => { const date = c.release_date || c.first_air_date || ''; return parseInt(date.slice(0, 4)) || 0; };
  const sortFn = (a, b) => yearOf(b) - yearOf(a) || (b.popularity || 0) - (a.popularity || 0);
  const movies = filtered.filter(c => c.media_type === 'movie'); movies.sort(sortFn);
  const tvs = filtered.filter(c => c.media_type === 'tv'); tvs.sort(sortFn);
  
  if (!ctx.filmTab) ctx.filmTab = movies.length > 0 ? 'movies' : 'tv';
  const wrap = el('div');
  
  const subTabs = el('div', 'card-tabs');
  const mTab = el('button', 'card-tab' + (ctx.filmTab === 'movies' ? ' active' : ''), `🎬 Фільми (${movies.length})`);
  const tTab = el('button', 'card-tab' + (ctx.filmTab === 'tv' ? ' active' : ''), `📺 Серіали (${tvs.length})`);
  mTab.addEventListener('click', () => { if (ctx.filmTab === 'movies') return; ctx.filmTab = 'movies'; renderCard(); });
  tTab.addEventListener('click', () => { if (ctx.filmTab === 'tv') return; ctx.filmTab = 'tv'; renderCard(); });
  subTabs.append(mTab, tTab);
  wrap.append(subTabs);
  
  const items = ctx.filmTab === 'movies' ? movies : tvs;
  if (!items.length) { wrap.append(el('div', 'avail-empty', 'Немає даних')); return [wrap]; }
  
  const grid = el('div', 'filmography-grid');
  grid.style.marginTop = '16px';
  for (const credit of items.slice(0, 100)) {
    const c = el('div', 'filmography-card');
    const poster = el('img', 'filmography-poster');
    poster.src = TMDB_IMG + 'w185' + credit.poster_path;
    poster.loading = 'lazy';
    c.append(poster);
    c.append(el('div', 'filmography-title', credit.title || credit.name || ''));
    const year = yearOf(credit);
    if (year) c.append(el('div', 'filmography-year', year));
    c.addEventListener('click', () => {
      const item = normalizeItem({ ...credit, media_type: credit.media_type });
      openMediaFromStack(item);
    });
    grid.append(c);
  }
  wrap.append(grid);
  return [wrap];
}

function renderPersonGallery(ctx) {
  if (!ctx.images || ctx.images.length === 0) return [el('div', 'avail-empty', 'Фотографій немає')];
  const gallery = el('div', 'photo-gallery');
  ctx.images.slice(0, 30).forEach(img => {
    const imgEl = el('img', 'gallery-photo');
    imgEl.src = TMDB_IMG + 'w185' + img.file_path;
    imgEl.loading = 'lazy';
    imgEl.alt = '';
    imgEl.addEventListener('click', () => openImageZoom(img.file_path, 'w500'));
    gallery.append(imgEl);
  });
  return [gallery];
}

function normalizeUrl(url) {
  if (!url) return '';
  let u = url.trim();
  if (!u.startsWith('http://') && !u.startsWith('https://')) {
    u = 'https://' + u;
  }
  return u.replace(/\/$/, '');
}

async function openTorrentSearch(item) {
  isCancelled = false;
  let jUrl = localStorage.getItem('mycinema.jackett_url') || 'https://jac-red.ru';
  const jKey = localStorage.getItem('mycinema.jackett_key') || 'viewbox';
  jUrl = normalizeUrl(jUrl);

  const lModal = document.getElementById('loading-modal');
  const lPoster = document.getElementById('loading-poster');
  const lStatus = document.getElementById('loading-status');
  const lSub = document.getElementById('loading-substatus');
  
  lPoster.src = TMDB_IMG + 'w500' + (item.poster || item.backdrop || '');
  lPoster.style.filter = 'grayscale(100%)';
  lStatus.textContent = 'Шукаю торенти...';
  lSub.textContent = 'Запит до Jackett';
  lModal.style.display = 'flex';
  
  try {
    const q = encodeURIComponent(item.title + (item.year ? ' ' + item.year : ''));
    const res = await fetch(`${jUrl}/api/v2.0/indexers/all/results?apikey=${jKey}&Query=${q}`);
    const data = await res.json();
    
    if (isCancelled) return;

    let results = (data.Results || []).filter(r => r.MagnetUri);
    results.sort((a, b) => (b.Seeders || 0) - (a.Seeders || 0));
    
    if (results.length === 0) {
      lModal.style.display = 'none';
      alert('Торентів не знайдено.');
      return;
    }

    lPoster.style.filter = 'grayscale(0%)';
    setTimeout(() => {
      if (isCancelled) return;
      lModal.style.display = 'none';
      showTorrentsList(results, item);
    }, 1500);

  } catch (e) {
    lModal.style.display = 'none';
    if (!isCancelled) alert('Помилка пошуку: ' + e.message);
  }
}

function parseEpisode(filePath) {
  const m1 = filePath.match(/S(\d+)E(\d+)/i);
  if (m1) return { season: parseInt(m1[1]), episode: parseInt(m1[2]) };
  const m2 = filePath.match(/(\d+)x(\d+)/i);
  if (m2) return { season: parseInt(m2[1]), episode: parseInt(m2[2]) };
  const m3 = filePath.match(/E(\d+)/i);
  if (m3) return { season: 1, episode: parseInt(m3[1]) };
  return { season: 0, episode: 0 };
}

function showFilesList(files, hash, item, tUrl, extPlayer) {
  const modal = document.getElementById('files-modal');
  const list = document.getElementById('files-list');
  
  if (!modal) {
      alert('Помилка: не знайдено вікно files-modal. Перевірте index.html!');
      return;
  }
  
  list.innerHTML = '';

  const sorted = [...files].sort((a, b) => {
    const ea = parseEpisode(a.path);
    const eb = parseEpisode(b.path);
    return (ea.season - eb.season) || (ea.episode - eb.episode);
  });

  const saved = lib[item.key] || {};
  const watchedEps = new Set(saved.watchedEpisodes || []);

  sorted.forEach(f => {
    const ep = parseEpisode(f.path);
    const fileName = f.path.split('/').pop();
    const sizeMB = (f.length / (1024 * 1024)).toFixed(0);
    const isWatched = ep.episode > 0 && watchedEps.has(`${ep.season}-${ep.episode}`);

    let epLabel = '▶';
    if (ep.episode > 0) {
      epLabel = `S${String(ep.season).padStart(2,'0')}E${String(ep.episode).padStart(2,'0')}`;
    }

    const div = document.createElement('div');
    div.className = 'file-item' + (isWatched ? ' watched' : '');

    div.innerHTML = `
      <div class="file-num">${epLabel}</div>
      <div class="file-name">${fileName}</div>
      <div class="file-size">${sizeMB} MB</div>
      ${isWatched ? '<div class="file-check">✓</div>' : ''}
    `;

    div.addEventListener('click', () => {
      closeFilesModal();
      playStream(f, hash, item, tUrl, extPlayer, ep.season, ep.episode);
    });

    list.append(div);
  });

  modal.style.display = 'flex';
}

function closeFilesModal() {
  const modal = document.getElementById('files-modal');
  if (modal) modal.style.display = 'none';
}

function playStream(videoFile, hash, item, tUrl, extPlayer, season, episode) {
  const ext = (videoFile.path.match(/\.\w+$/) || ['.mkv'])[0];
  const streamUrl = `${tUrl}/stream/video${ext}?link=${hash}&index=${videoFile.id}&play`;
  console.log("Stream URL:", streamUrl);

  if (season > 0 && episode > 0) {
    if (!lib[item.key]) {
      lib[item.key] = { ...item, status: 'watching', watchedEpisodes: [], updated: Date.now() };
    }
    if (!lib[item.key].watchedEpisodes) lib[item.key].watchedEpisodes = [];
    
    const epKey = `${season}-${episode}`;
    if (!lib[item.key].watchedEpisodes.includes(epKey)) {
      lib[item.key].watchedEpisodes.push(epKey);
      if (lib[item.key].status !== 'watched') lib[item.key].status = 'watching';
      lib[item.key].updated = Date.now();
      save();
      showSyncToast(`✅ S${String(season).padStart(2,'0')}E${String(episode).padStart(2,'0')} позначено`);
    }
  }

  if (!extPlayer) {
    window.open(streamUrl, '_blank');
    return;
  }

  if (!window.Capacitor || !window.Capacitor.Plugins || !window.Capacitor.Plugins.ExternalPlayer) {
    alert('Плагін ExternalPlayer не знайдено!');
    return;
  }

  const plugin = window.Capacitor.Plugins.ExternalPlayer;

  plugin.open({
    url: streamUrl,
    title: item.title || 'MyCinema'
  }).then(r => {
    console.log('Відкрито через:', r.via);
  }).catch(err => {
    alert('Помилка плеєра: ' + (err.message || JSON.stringify(err)));
  });
}

async function playTorrent(magnet, item) {
  isCancelled = false;
  const tUrl = normalizeUrl(localStorage.getItem('mycinema.torrserver_url') || 'http://127.0.0.1:8090');
  const extPlayer = localStorage.getItem('mycinema.external_player') !== 'false';
  
  const lModal = document.getElementById('loading-modal');
  const lPoster = document.getElementById('loading-poster');
  const lStatus = document.getElementById('loading-status');
  const lSub = document.getElementById('loading-substatus');
  
  lPoster.src = TMDB_IMG + 'w500' + (item.poster || item.backdrop || '');
  lPoster.style.filter = 'grayscale(0%)';
  lStatus.textContent = 'Підготовка стріму...';
  lSub.textContent = 'TorrServer аналізує файли';
  lModal.style.display = 'flex';

  try {
    lSub.textContent = 'Додаю торрент...';
    const addRes = await fetch(`${tUrl}/torrents`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ action: 'add', link: magnet, save_to_db: true })
    });

    if (isCancelled) return;

    const addText = await addRes.text();
    let hash = null;
    let files = [];

    try {
        const data = JSON.parse(addText);
        if (data && data.hash) {
            hash = data.hash;
            if (data.file_stats && Array.isArray(data.file_stats)) {
                files = data.file_stats.filter(f => f.path && f.path.match(/\.(mkv|mp4|avi)$/i));
            }
        }
    } catch (e) {}

    if (!hash) {
        const hashMatch = magnet.match(/btih:([a-f0-9]{40})/i);
        if (hashMatch) hash = hashMatch[1];
    }

    if (!hash) throw new Error('Не вдалося отримати хеш');

    let attempts = 0;
    while (files.length === 0 && attempts < 15) { 
      if (isCancelled) return;
      
      lSub.textContent = `Отримую файли... (спроба ${attempts + 1})`;
      const statRes = await fetch(`${tUrl}/torrents/${hash}`);
      if (statRes.ok) {
          const statText = await statRes.text();
          try {
              const statData = JSON.parse(statText);
              if (statData && statData.file_stats) {
                  files = statData.file_stats.filter(f => f.path && f.path.match(/\.(mkv|mp4|avi)$/i));
              }
          } catch (e) {}
      }
      if (files.length === 0) {
          await new Promise(r => setTimeout(r, 2000));
          attempts++;
      }
    }

    if (isCancelled) return;
    if (files.length === 0) throw new Error('Відеофайлів не знайдено');

    lModal.style.display = 'none';

    files.sort((a, b) => b.length - a.length);

    const hasEpisodes = files.some(f => /S\d+E\d+/i.test(f.path));
    const isTV = item.type === 'tv';

    if (files.length > 1 && (isTV || hasEpisodes)) {
      showFilesList(files, hash, item, tUrl, extPlayer);
    } else {
      playStream(files[0], hash, item, tUrl, extPlayer, 0, 0);
    }

  } catch (e) {
    lModal.style.display = 'none';
    if (!isCancelled) alert('Помилка TorrServer: ' + e.message);
  }
}

function renderTasteTab() {
  const taste = computeTasteVector();
  if (!taste || taste.count < 3) return [el('div', 'avail-empty', 'Потрібно мінімум 3 оцінки, щоб побачити аналіз')];
  const wrap = el('div');

  wrap.append(buildTasteCard(taste));

  const cutoff = Date.now() - 30 * 24 * 60 * 60 * 1000;
  const recent = Object.values(lib).filter(x => (x.updated || 0) > cutoff && computeItemRating(x).avg > 0);
  
  if (recent.length >= 3) {
    const recentTaste = { count: recent.length };
    const sums = {}, weights = {};
    for (const crit of RATING_CRITERIA) { sums[crit.key] = 0; weights[crit.key] = 0; }
    for (const item of recent) {
      const rr = item.type === 'tv' ? averagedSeasonRatings(item) : (item.ratings || {});
      for (const crit of RATING_CRITERIA) {
        if (rr[crit.key]) { sums[crit.key] += rr[crit.key]; weights[crit.key]++; }
      }
    }
    for (const crit of RATING_CRITERIA) recentTaste[crit.key] = weights[crit.key] > 0 ? sums[crit.key] / weights[crit.key] : 0;

    wrap.append(el('h3', 'label', '📈 Еволюція · 30 днів vs весь час'));
    const trend = el('div', 'trend-chart');
    for (const crit of RATING_CRITERIA) {
      const oldV = taste[crit.key] || 0;
      const newV = recentTaste[crit.key] || 0;
      const diff = newV - oldV;
      const row = el('div', 'trend-item');
      const labels = el('div', 'trend-labels');
      labels.append(el('span', '', crit.label));
      const sign = diff > 0.1 ? '↑' : (diff < -0.1 ? '↓' : '→');
      const color = diff > 0.1 ? 'var(--mint)' : (diff < -0.1 ? 'var(--danger)' : 'var(--text-muted)');
      const right = el('span'); right.innerHTML = `${fmtNum(oldV)} → <strong style="color:${color}">${fmtNum(newV)}</strong> ${sign}`;
      labels.append(right);
      row.append(labels);
      const bars = el('div', 'trend-bars');
      const oldBar = el('div', 'trend-bar-old'); oldBar.style.width = (oldV / 10 * 100) + '%';
      const newBar = el('div', 'trend-bar-new'); newBar.style.width = (newV / 10 * 100) + '%';
      bars.append(oldBar, newBar);
      row.append(bars);
      trend.append(row);
    }
    wrap.append(trend);
    const legend = el('div', 'trend-legend');
    legend.innerHTML = '<span class="old">весь час</span><span class="new">30 днів</span>';
    wrap.append(legend);
  }

  wrap.append(el('h3', 'label', '💡 Інсайти'));
  const insights = [];

  const rated = Object.values(lib).map(x => ({ item: x, r: computeItemRating(x) })).filter(x => x.r.avg > 0 && x.item.ratings);
  if (rated.length >= 5) {
    const plotHigh = rated.filter(x => (x.item.ratings?.plot || 0) >= 8);
    const castHigh = rated.filter(x => (x.item.ratings?.cast || 0) >= 8);
    if (plotHigh.length >= 3 && castHigh.length >= 3) {
      const overlap = plotHigh.filter(x => (x.item.ratings?.cast || 0) >= 8).length;
      if (overlap / plotHigh.length > 0.6) {
        insights.push({ icon: '🎯', text: `Коли тобі подобається сюжет (<strong>8+</strong>), ти майже завжди високо оцінюєш і каст. У <strong>${Math.round(overlap / plotHigh.length * 100)}%</strong> випадків.` });
      }
    }
  }

  const allRated = rated.map(x => x.r.avg);
  if (allRated.length >= 5) {
    const avg = allRated.reduce((a, b) => a + b, 0) / allRated.length;
    const strict = avg < 6.5, generous = avg > 7.5;
    insights.push({ icon: generous ? '🌟' : (strict ? '🔍' : '⚖️'), text: `Твоя середня оцінка — <strong>${fmtNum(avg)}/10</strong>. ${generous ? 'Ти щедрий на оцінки — ставиш високі бали частіше, ніж інші.' : strict ? 'Ти вимогливий — не роздаєш високі оцінки просто так.' : 'Ти ставишся до оцінювання виважено.'}` });
  }

  if (rated.length > 0) {
    const top = rated.sort((a, b) => b.r.avg - a.r.avg)[0];
    insights.push({ icon: '👑', text: `Твій найвищий бал — <strong>${fmtNum(top.r.avg)}/10</strong> — у «${top.item.title}».` });
  }

  const movieRated = rated.filter(x => x.item.type === 'movie');
  const tvRated = rated.filter(x => x.item.type === 'tv');
  if (movieRated.length >= 3 && tvRated.length >= 3) {
    const mAvg = movieRated.reduce((a, b) => a + b.r.avg, 0) / movieRated.length;
    const tAvg = tvRated.reduce((a, b) => a + b.r.avg, 0) / tvRated.length;
    const diff = tAvg - mAvg;
    if (Math.abs(diff) > 0.5) {
      insights.push({ icon: diff > 0 ? '📺' : '🎬', text: `Ти оцінюєш ${diff > 0 ? 'серіали' : 'фільми'} у середньому на <strong>${fmtNum(Math.abs(diff))}</strong> вище, ніж ${diff > 0 ? 'фільми' : 'серіали'}.` });
    }
  }

  if (insights.length) {
    for (const ins of insights) {
      const c = el('div', 'insight-card');
      c.append(el('div', 'insight-label', ins.icon + ' інсайт'));
      const t = el('div', 'insight-text');
      t.innerHTML = ins.text;
      c.append(t);
      wrap.append(c);
    }
  }

  return [wrap];
}

function renderActorsTab() {
  const wrap = el('div');
  const subTabs = el('div', 'card-tabs');
  const rTab = el('button', 'card-tab' + (state.stats.actorTab === 'rating' ? ' active' : ''), '⭐ За оцінкою');
  const cTab = el('button', 'card-tab' + (state.stats.actorTab === 'count' ? ' active' : ''), '👁️ За переглядами');
  rTab.addEventListener('click', () => { if (state.stats.actorTab === 'rating') return; state.stats.actorTab = 'rating'; renderStatistics(); });
  cTab.addEventListener('click', () => { if (state.stats.actorTab === 'count') return; state.stats.actorTab = 'count'; renderStatistics(); });
  subTabs.append(rTab, cTab);
  wrap.append(subTabs);
  const content = el('div'); content.style.marginTop = '16px';
  const items = Object.values(lib);
  const missing = items.filter(x => !x.cast || !x.cast.length);
  
  if (missing.length > 0) {
    const loader = el('div', 'cast-loader');
    if (state.stats.castLoading) {
      loader.append(el('div', 'cast-loader-text', `Завантажую акторів: ${state.stats.castProgress.done}/${state.stats.castProgress.total}...`));
    } else {
      loader.append(el('div', 'cast-loader-text', `Даних про акторів немає для ${missing.length} з ${items.length} тайтлів`));
      const btn = el('button', 'cast-loader-btn', '📥 Завантажити дані акторів');
      btn.addEventListener('click', async () => {
        state.stats.castLoading = true;
        state.stats.castProgress = { done: 0, total: missing.length };
        renderStatistics();
        await fetchAllMissingCasts();
        state.stats.castLoading = false;
        renderStatistics();
      });
      loader.append(btn);
    }
    wrap.append(loader);
  }
  
  const actorMap = new Map();
  items.forEach(item => {
    (item.cast || []).forEach(a => {
      if (!actorMap.has(a.id)) actorMap.set(a.id, { id: a.id, name: a.name, profile_path: a.profile_path, projects: [] });
      actorMap.get(a.id).projects.push(item);
    });
  });

  const allRatedAvgs = Object.values(lib).map(x => computeItemRating(x).avg).filter(v => v > 0);
  const globalAvg = allRatedAvgs.length ? allRatedAvgs.reduce((a, b) => a + b, 0) / allRatedAvgs.length : 7;

  const actors = [...actorMap.values()].map(a => {
    const watched = a.projects.filter(hasWatched);
    const rated = watched.filter(p => computeItemRating(p).avg > 0);
    const avgRating = rated.length ? rated.reduce((s, p) => s + computeItemRating(p).avg, 0) / rated.length : 0;
    const v = rated.length;
    const weightedRating = v > 0 ? (v * avgRating + ACTOR_MIN_VOTES * globalAvg) / (v + ACTOR_MIN_VOTES) : 0;
    return { ...a, watchedCount: watched.length, ratedCount: v, avgRating, weightedRating };
  });

  let display, valueFn;
  if (state.stats.actorTab === 'rating') {
    display = actors.filter(a => a.ratedCount >= 1).sort((x, y) => y.weightedRating - x.weightedRating || y.ratedCount - x.ratedCount || x.name.localeCompare(y.name));
    valueFn = a => ({ text: fmtNum(a.weightedRating), cls: ratingClass(a.weightedRating) });
  } else {
    display = actors.filter(a => a.watchedCount >= 1).sort((x, y) => y.watchedCount - x.watchedCount || y.weightedRating - x.weightedRating || x.name.localeCompare(y.name));
    valueFn = a => ({ text: String(a.watchedCount), cls: '' });
  }

  if (!display.length) {
    content.append(el('div', 'avail-empty', state.stats.actorTab === 'rating' ? 'Немає оцінених проєктів з акторами' : 'Немає переглянутих проєктів з акторами'));
    wrap.append(content);
    return [wrap];
  }

  if (state.stats.actorTab === 'rating' && display.length > 0) {
    const hint = el('div');
    hint.style.cssText = 'padding:10px 14px;background:rgba(255,184,108,.06);border-left:3px solid var(--accent);border-radius:8px;font-size:12px;color:var(--text-muted);line-height:1.5;margin-bottom:12px';
    hint.innerHTML = `Оцінка зважена за кількістю проєктів (мін. вага: <strong style="color:var(--accent)">${ACTOR_MIN_VOTES}</strong>). Актори з малою кількістю оцінок тягнуться до середнього (<strong style="color:var(--accent)">${fmtNum(globalAvg)}</strong>), тому не "висять" у топі випадково.`;
    content.append(hint);
  }

  const list = el('div', 'actor-rank-list');
  display.slice(0, 50).forEach((a, idx) => {
    const it = el('div', 'actor-rank-item');
    it.append(el('div', 'actor-rank-num', String(idx + 1)));
    if (a.profile_path) {
      const ph = el('img', 'actor-rank-photo'); ph.src = TMDB_IMG + 'w185' + a.profile_path; ph.loading = 'lazy';
      it.append(ph);
    } else {
      it.append(el('div', 'actor-rank-photo-empty', (a.name || '?').charAt(0).toUpperCase()));
    }
    const info = el('div', 'actor-rank-info');
    info.append(el('div', 'actor-rank-name', a.name));
    let meta;
    if (state.stats.actorTab === 'rating') {
      meta = `${a.ratedCount} ${plural(a.ratedCount, 'проєкт', 'проєкти', 'проєктів')} · сира ${fmtNum(a.avgRating)}`;
    } else {
      meta = `${a.watchedCount} ${plural(a.watchedCount, 'перегляд', 'перегляди', 'переглядів')}`;
    }
    info.append(el('div', 'actor-rank-meta', meta));
    it.append(info);
    const v = valueFn(a);
    it.append(el('div', 'actor-rank-value ' + v.cls, v.text));
    it.addEventListener('click', () => openPerson(a.id, { name: a.name }));
    list.append(it);
  });
  content.append(list);
  wrap.append(content);
  return [wrap];
}

async function fetchAllMissingCasts() {
  const missing = Object.values(lib).filter(x => !x.cast || !x.cast.length);
  const batchSize = 5;
  for (let i = 0; i < missing.length; i += batchSize) {
    const batch = missing.slice(i, i + batchSize);
    await Promise.all(batch.map(async item => {
      try {
        const d = await tmdbFetch(`/${item.type}/${item.id}/credits?language=uk-UA`);
        if (lib[item.key] && d.cast) lib[item.key].cast = d.cast.slice(0, CAST_LIMIT).map(a => ({ id: a.id, name: a.name, profile_path: a.profile_path }));
      } catch {}
    }));
    state.stats.castProgress.done = Math.min(missing.length, i + batchSize);
    if (state.stats.tab === 'actors') renderStatistics();
    await new Promise(r => setTimeout(r, 200));
  }
  save();
}

function renderDetailTab() {
  const all = Object.values(lib);
  if (!all.length) return [el('div', 'avail-empty', 'Бібліотека порожня')];
  const wrap = el('div');

  const movies = all.filter(x => x.type === 'movie').length;
  const tvShows = all.filter(x => x.type === 'tv').length;
  const typeData = [
    { label: 'Фільми', value: movies, color: 'var(--accent)' },
    { label: 'Серіали', value: tvShows, color: 'var(--violet)' }
  ].filter(d => d.value > 0);
  wrap.append(renderPieSection('🎬 Тип контенту', typeData));

  const statuses = ['watched', 'watching', 'planned', 'dropped'];
  const labels = { watched: 'Переглянуто', watching: 'Дивлюсь', planned: 'Планую', dropped: 'Закинуто' };
  const colors = { watched: 'var(--mint)', watching: 'var(--accent)', planned: 'var(--violet)', dropped: 'var(--text-muted)' };
  const statusData = statuses.map(s => ({ label: labels[s], value: all.filter(x => x.status === s).length, color: colors[s] })).filter(d => d.value > 0);
  wrap.append(renderPieSection('📌 Статуси', statusData));

  const ratedMovies = all.filter(x => x.type === 'movie' && computeItemRating(x).avg > 0);
  const ratedTV = all.filter(x => x.type === 'tv' && computeItemRating(x).avg > 0);
  const avgMovie = ratedMovies.length ? ratedMovies.reduce((s, x) => s + computeItemRating(x).avg, 0) / ratedMovies.length : 0;
  const avgTV = ratedTV.length ? ratedTV.reduce((s, x) => s + computeItemRating(x).avg, 0) / ratedTV.length : 0;

  const sg = el('div', 'stats-grid');
  const addStat = (l, v) => { const s = el('div', 'stat-card'); s.append(el('div', 'stat-value', String(v))); s.append(el('div', 'stat-label', l)); sg.append(s); };
  addStat('Всього', all.length);
  addStat('Оцінено', ratedMovies.length + ratedTV.length);
  addStat('Сер. фільми', avgMovie > 0 ? fmtNum(avgMovie) : '—');
  addStat('Сер. серіали', avgTV > 0 ? fmtNum(avgTV) : '—');
  wrap.append(sg);

  const dist = {};
  for (let i = 1; i <= 10; i++) dist[i] = 0;
  [...ratedMovies, ...ratedTV].forEach(x => {
    const val = Math.round(computeItemRating(x).avg);
    if (val >= 1 && val <= 10) dist[val]++;
  });
  const maxDist = Math.max(...Object.values(dist), 1);
  const hist = el('div', 'histogram');
  for (let i = 10; i >= 1; i--) {
    if (dist[i] === 0) continue;
    const row = el('div', 'histogram-row');
    row.append(el('div', 'histogram-label', i + '/10'));
    const barBg = el('div', 'histogram-bar-bg');
    const bar = el('div', 'histogram-bar');
    bar.style.width = ((dist[i] / maxDist) * 100) + '%';
    bar.textContent = dist[i];
    barBg.append(bar);
    row.append(barBg);
    hist.append(row);
  }
  if (hist.children.length) {
    wrap.append(el('h3', 'label', '📈 Розподіл оцінок'));
    wrap.append(hist);
  }
  return [wrap];
}

function renderPieSection(title, data) {
  const section = el('div', 'pie-section');
  section.append(el('div', 'pie-title', title));
  const content = el('div', 'pie-content');
  content.append(renderPieChart(data));
  const legend = el('div', 'pie-legend');
  const total = data.reduce((s, d) => s + d.value, 0);
  if (total === 0) { section.append(el('div', 'avail-empty', 'Немає даних')); return section; }
  data.forEach(d => {
    const row = el('div', 'pie-legend-row');
    const dot = el('div', 'pie-legend-dot'); dot.style.background = d.color;
    row.append(dot, el('span', 'pie-legend-label', d.label));
    row.append(el('span', 'pie-legend-value', `${d.value} · ${Math.round(d.value / total * 100)}%`));
    legend.append(row);
  });
  content.append(legend);
  section.append(content);
  return section;
}

function renderPieChart(data, size = 160) {
  const total = data.reduce((s, d) => s + d.value, 0);
  if (total === 0) return el('div');
  const svgNS = 'http://www.w3.org/2000/svg';
  const svg = document.createElementNS(svgNS, 'svg');
  svg.setAttribute('viewBox', `0 0 ${size} ${size}`);
  svg.setAttribute('width', size);
  svg.setAttribute('height', size);
  svg.setAttribute('class', 'pie-svg');
  let angle = -Math.PI / 2;
  const cx = size / 2, cy = size / 2, r = size / 2 - 4;
  data.forEach(d => {
    if (d.value === 0) return;
    const sweep = (d.value / total) * Math.PI * 2;
    const x1 = cx + r * Math.cos(angle);
    const y1 = cy + r * Math.sin(angle);
    const x2 = cx + r * Math.cos(angle + sweep);
    const y2 = cy + r * Math.sin(angle + sweep);
    const largeArc = sweep > Math.PI ? 1 : 0;
    const path = document.createElementNS(svgNS, 'path');
    path.setAttribute('d', `M ${cx} ${cy} L ${x1} ${y1} A ${r} ${r} 0 ${largeArc} 1 ${x2} ${y2} Z`);
    path.setAttribute('fill', d.color);
    path.setAttribute('stroke', 'var(--bg-0)');
    path.setAttribute('stroke-width', '2');
    svg.appendChild(path);
    angle += sweep;
  });
  const circle = document.createElementNS(svgNS, 'circle');
  circle.setAttribute('cx', cx); circle.setAttribute('cy', cy);
  circle.setAttribute('r', r * 0.55);
  circle.setAttribute('fill', 'var(--bg-1)');
  svg.appendChild(circle);
  const text = document.createElementNS(svgNS, 'text');
  text.setAttribute('x', cx); text.setAttribute('y', cy + 6);
  text.setAttribute('text-anchor', 'middle');
  text.setAttribute('fill', 'var(--text)');
  text.setAttribute('font-size', '18');
  text.setAttribute('font-weight', '800');
  text.setAttribute('font-family', 'var(--mono)');
  text.textContent = String(total);
  svg.appendChild(text);
  return svg;
}

  const ripple = document.createElement('span');
  ripple.className = 'ripple';
  ripple.style.width = size + 'px';
  ripple.style.height = size + 'px';

  ripple.style.top = (e.clientY - rect.top - size / 2) + 'px';
